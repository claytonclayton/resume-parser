\newcommand{\Dot}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

\NewDocumentCommand{\Block}{m O{} O{} O{} O{}}{
    \vspace{0pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \IfBlankTF{#2}
      {\textbf{#1} & \textit{\small #3} \\}
      {\textbf{#1} $|$ \textit{\small#2} & \textit{\small #3} \\}
      \IfBlankF{#4}
      {\textit{\small#4} & \textit{\small #5} \\}
    \end{tabular*}
    \vspace{-2pt}
}

\NewDocumentCommand{\SubBlock}{m O{}}{
    \item
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2}
    \end{tabular*}
}
